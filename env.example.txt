# =============================================================================
# MQTT API Gateway Configuration
# =============================================================================
# Rename this file to .env and update the values for your environment.

# -----------------------------------------------------------------------------
# MQTT Broker Settings
# -----------------------------------------------------------------------------
# Hostname or IP address of your MQTT broker
MQTT_BROKER_HOST=your-broker.cloud.com

# Port number (typically 8883 for TLS, 1883 for non-TLS)
MQTT_BROKER_PORT=8883

# MQTT authentication credentials
MQTT_USERNAME=your_mqtt_username
MQTT_PASSWORD=your_mqtt_password

# Enable TLS/SSL encryption (true/false)
MQTT_USE_TLS=true

# -----------------------------------------------------------------------------
# MQTT Topics
# -----------------------------------------------------------------------------
# Comma-separated list of topic patterns to subscribe to
# IMPORTANT: Wrap in quotes to allow # wildcard characters!
# Use # as wildcard for all subtopics, + for single level wildcard
# Examples:
#   MQTT_TOPICS="sensors/#"
#   MQTT_TOPICS="factory/line1/#,factory/line2/#"
#   MQTT_TOPICS="#"  (subscribe to everything)
MQTT_TOPICS="Root/Site1/#,Root/Site2/#"

# -----------------------------------------------------------------------------
# API Authentication
# -----------------------------------------------------------------------------
# Credentials for HTTP Basic Authentication on the REST API
# These will be used by the REST API
API_USERNAME=user
API_PASSWORD=your_secure_api_password

# -----------------------------------------------------------------------------
# Memory Management (Optional - defaults shown)
# -----------------------------------------------------------------------------
# Maximum number of messages to store per topic
# Older messages are automatically evicted when this limit is reached
MAX_MESSAGES_PER_TOPIC=10000

# Number of hours to retain messages before automatic cleanup
HISTORY_RETENTION_HOURS=24

# How often (in minutes) to run the cleanup task
CLEANUP_INTERVAL_MINUTES=5
